CONTAINER_URNS=$(curl -sS "$DATAHUB_GMS_URL/api/graphql" \
  -H "Content-Type: application/json" \
  -H "Authorization: Bearer $DATAHUB_TOKEN" \
  -d '{"query":"query($q: String!){ search(input:{ type: CONTAINER, query: $q, start:0, count:50 }) { searchResults { entity { urn } } } }","variables":{"q":"classicmodels"}}' \
| python3 -c 'import sys,json; d=json.load(sys.stdin); print("\n".join(r["entity"]["urn"] for r in d["data"]["search"]["searchResults"]))')

# 2 Hapus masing-masing container (database/schema)

    -H "Content-Type: application/json" \
